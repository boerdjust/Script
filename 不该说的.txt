local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
   Name = "band's arsenal script",
   LoadingTitle = "Loading...",
   LoadingSubtitle = "by: band",
   ConfigurationSaving = {
      Enabled = true,
      FolderName = "AimbotStalkHub",
      FileName = "Config"
   },
   Discord = {
      Enabled = false,
      Invite = "noinv",
      RememberJoins = true
   },
   KeySystem = false
})

local Tab = Window:CreateTab("主要功能", 4483362458)

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local Camera = Workspace.CurrentCamera
local LocalPlayer = Players.LocalPlayer

local aimbotEnabled = false
local aimbotConnection = nil
local ignoreTeam = true
local smoothness = 0.12
local fovRadius = 120
local aimPart = "身體"

local selectedStalkPlayer = nil
local stalkEnabled = false
local stalkConnection = nil
local playerList = {}

local function updatePlayerList()
   playerList = {}
   for _, player in pairs(Players:GetPlayers()) do
      if player \~= LocalPlayer then
         table.insert(playerList, player.Name)
      end
   end
end
updatePlayerList()

Tab:CreateToggle({
   Name = "X",
   CurrentValue = false,
   Flag = "AimbotToggle",
   Callback = function(Value)
      aimbotEnabled = Value
      if Value then
         aimbotConnection = RunService.RenderStepped:Connect(function()
            local closestPlayer = nil
            local shortestDistance = math.huge
            
            local origin = Camera.CFrame.Position
            local viewportSize = Camera.ViewportSize
            local centerScreen = Vector2.new(viewportSize.X / 2, viewportSize.Y / 2)
            
            for _, player in pairs(Players:GetPlayers()) do
               if player \~= LocalPlayer and player.Character then
                  local humanoid = player.Character:FindFirstChild("Humanoid")
                  if humanoid and humanoid.Health > 0 then
                     if not ignoreTeam or player.Team \~= LocalPlayer.Team then
                        local targetPart
                        if aimPart == "頭部" then
                           targetPart = player.Character:FindFirstChild("Head")
                        else
                           targetPart = player.Character:FindFirstChild("UpperTorso") 
                                      or player.Character:FindFirstChild("Torso") 
                                      or player.Character:FindFirstChild("HumanoidRootPart")
                        end
                        
                        if targetPart then
                           local targetPos = targetPart.Position
                           local screenPos, onScreen = Camera:WorldToViewportPoint(targetPos)
                           
                           if onScreen then
                              local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - centerScreen).Magnitude
                              if screenDist < fovRadius then
                                 local distance = (targetPos - origin).Magnitude
                                 
                                 if distance < shortestDistance then
                                    local direction = (targetPos - origin)
                                    local raycastParams = RaycastParams.new()
                                    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
                                    raycastParams.FilterDescendantsInstances = {LocalPlayer.Character}
                                    
                                    local raycastResult = Workspace:Raycast(origin, direction.Unit * distance, raycastParams)
                                    
                                    if not raycastResult or raycastResult.Instance:IsDescendantOf(player.Character) then
                                       local rootPart = player.Character:FindFirstChild("HumanoidRootPart")
                                       local velocity = rootPart and rootPart.Velocity or Vector3.new()
                                       local predictedPos = targetPos + velocity * 0.1
                                       
                                       closestPlayer = {player = player, pos = predictedPos}
                                       shortestDistance = distance
                                    end
                                 end
                              end
                           end
                        end
                     end
                  end
               end
            end
            
            if closestPlayer then
               local targetCFrame = CFrame.lookAt(origin, closestPlayer.pos)
               Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, smoothness)
            end
         end)
         
         Rayfield:Notify({
            Title = "自動瞄準已啟用",
            Content = "鎖定最近可視玩家 (" .. aimPart .. ")",
            Duration = 3,
            Image = 4483362458
         })
      else
         if aimbotConnection then
            aimbotConnection:Disconnect()
            aimbotConnection = nil
         end
      end
   end,
})

Tab:CreateToggle({
   Name = "忽略隊友",
   CurrentValue = true,
   Flag = "IgnoreTeam",
   Callback = function(Value)
      ignoreTeam = Value
   end,
})

Tab:CreateDropdown({
   Name = "瞄準部位",
   Options = {"身體", "頭部"},
   CurrentOption = {"身體"},
   Flag = "AimPart",
   Callback = function(Option)
      aimPart = Option[1]
   end,
})

Tab:CreateSlider({
   Name = "瞄準平滑度 (越低越自然)",
   Range = {5, 50},
   Increment = 5,
   Suffix = "%",
   CurrentValue = 12,
   Flag = "Smoothness",
   Callback = function(Value)
      smoothness = Value / 100
   end,
})

Tab:CreateSlider({
   Name = "FOV 範圍",
   Range = {50, 300},
   Increment = 25,
   Suffix = "px",
   CurrentValue = 120,
   Flag = "FOV",
   Callback = function(Value)
      fovRadius = Value
   end,
})

Tab:CreateParagraph({
   Title = "────────── 尾隨功能 ──────────",
   Content = ""
})

Tab:CreateDropdown({
   Name = "選擇尾隨目標",
   Options = playerList,
   CurrentOption = {"無"},
   MultipleOptions = false,
   Flag = "StalkPlayer",
   Callback = function(Option)
      if Option[1] \~= "無" then
         selectedStalkPlayer = Players:FindFirstChild(Option[1])
         Rayfield:Notify({
            Title = "已選擇尾隨目標",
            Content = "將持續緊貼 " .. Option[1] .. " 後面",
            Duration = 3,
            Image = 4483362458
         })
      else
         selectedStalkPlayer = nil
      end
   end,
})

Tab:CreateButton({
   Name = "刷新目標列表",
   Callback = function()
      updatePlayerList()
      Tab.Flags.StalkPlayer:Refresh(playerList, true)
      Rayfield:Notify({
         Title = "列表已刷新",
         Content = "尾隨目標列表更新完成",
         Duration = 2,
         Image = 4483362458
      })
   end,
})

Tab:CreateToggle({
   Name = "啟用持續尾隨（緊貼目標後方）",
   CurrentValue = false,
   Flag = "StalkToggle",
   Callback = function(Value)
      stalkEnabled = Value
      
      if Value and selectedStalkPlayer then
         stalkConnection = RunService.Heartbeat:Connect(function()
            if selectedStalkPlayer.Character 
               and selectedStalkPlayer.Character:FindFirstChild("HumanoidRootPart")
               and LocalPlayer.Character 
               and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                
               local offset = CFrame.new(0, 0, 0.6)
               LocalPlayer.Character.HumanoidRootPart.CFrame = 
                  selectedStalkPlayer.Character.HumanoidRootPart.CFrame * offset
            end
         end)
         
         Rayfield:Notify({
            Title = "尾隨已啟用",
            Content = "已緊貼在 " .. selectedStalkPlayer.Name .. " 後方",
            Duration = 4,
            Image = 4483362458
         })
      else
         if stalkConnection then
            stalkConnection:Disconnect()
            stalkConnection = nil
         end
         Rayfield:Notify({
            Title = "尾隨已關閉",
            Content = "已停止尾隨",
            Duration = 3,
            Image = 4483362458
         })
      end
   end,
})

Players.PlayerAdded:Connect(function()
   task.wait(1)
   updatePlayerList()
   Tab.Flags.StalkPlayer:Refresh(playerList, true)
end)

Players.PlayerRemoving:Connect(function()
   task.wait(1)
   updatePlayerList()
   Tab.Flags.StalkPlayer:Refresh(playerList, true)
end)

Rayfield:Notify({
   Title = "腳本載入完成",
   Content = "版本: 1.1 | by band",
   Duration = 8,
   Image = 4483362458
})
